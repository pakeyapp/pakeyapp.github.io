# 1. Initialize a new Git repository in your project folder
git init -b main

# 2. Add your Codeberg repository as the remote origin
# (Replace 'pakeyapp' if you chose a different username)
git remote add origin https://github.com/pakeyapp/pakeyapp.github.io.git

# 3. Add all your files to be committed
git add .

# 4. Commit your files with a message
git commit -m "Initial website commit"

# 5. Push your files to Codeberg
git push -u origin main
